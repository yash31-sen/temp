class MarkdownProcessor:

EXTENSIONS = [
"extra",
"toc",
"tables",
"fenced_code",
"codehilite",
"sane_lists",
"smarty",
"attr_list",
"md_in_html",
"footnotes"
 ]

@staticmethod
def process(content: str):

md = markdown.Markdown(
extensions=MarkdownProcessor.EXTENSIONS,
extension_configs={
"toc": {
"title": "Contents",
"permalink": True,
"toc_depth": "1-4"
},
"codehilite": {
"guess_lang": True,
"linenums": False,
"css_class": "highlight"
}
},
output_format="html5"
)

rendered_html = md.convert(content)

# Generate TOC separately so it appears automatically,
# even if the original Markdown does not contain [TOC].
toc_html = md.toc

return rendered_html, toc_html

DOCUMENT METADATA

def extract_title(rendered_html: str, source: Path) -> str:

soup = BeautifulSoup(rendered_html, "html.parser")

heading = soup.find("h1")

if heading:
title = heading.get_text(" ", strip=True)
if title:
return title

return source.stem.replace("_", " ").replace("-", " ").title()

class SummaryGenerator:
"""
Creates a local extractive summary.

Selects important sentences based on word frequency.
It does not invent new facts or call an external AI service.
"""

@staticmethod
def generate(rendered_html: str, max_sentences: int = 5) -> list:

soup = BeautifulSoup(rendered_html, "html.parser")

# Exclude code, navigation, and non-text elements.
for element in soup.select(
"pre, code, script, style, nav, img, svg"
):
element.decompose()

text = soup.get_text(" ", strip=True)

# Remove excessive whitespace.
text = re.sub(r"\s+", " ", text).strip()

if not text:
return ["No readable textual content was found."]

# Split text into sentences.
sentences = re.split(
r'(?<=[.!?])\s+(?=[A-Z0-9])',
text
)

sentences = [
sentence.strip()
for sentence in sentences
if len(sentence.split()) >= 5
 ]

if not sentences:
return [text[:500]]

# Extract meaningful words.
words = re.findall(r"\b[a-zA-Z0-9_-]+\b", text.lower())

meaningful_words = [
word for word in words
if word not in STOP_WORDS and len(word) > 2
 ]

frequency = Counter(meaningful_words)

if not frequency:
return sentences[:max_sentences]

max_frequency = max(frequency.values())

# Normalize word frequencies.
scores = {
word: count / max_frequency
for word, count in frequency.items()
}

ranked = []

for index, sentence in enumerate(sentences):

sentence_words = re.findall(
r"\b[a-zA-Z0-9_-]+\b",
sentence.lower()
)

valid_words = [
word for word in sentence_words
if word not in STOP_WORDS and len(word) > 2
 ]

if not valid_words:
continue

score = sum(
scores.get(word, 0)
for word in valid_words
) / len(valid_words)

# Slightly prefer informative sentences,
# without allowing very long sentences to dominate.
score *= min(len(valid_words), 30) / 20

ranked.append((score, index, sentence))

ranked.sort(key=lambda item: item[0], reverse=True)

selected = ranked[:max_sentences]

# Restore original document order.
selected.sort(key=lambda item: item[1])

return [item[2] for item in selected]

class PDFConverter:

@staticmethod
def convert(markdown_path: Path, output_path: Path):

logger.info("Reading Markdown file: %s", markdown_path)

content = markdown_path.read_text(encoding="utf-8")

if not content.strip():
raise ValueError("The Markdown file is empty.")

logger.info("Processing Markdown content...")

rendered_html, toc_html = MarkdownProcessor.process(
content
)

title = extract_title(rendered_html, markdown_path)

logger.info("Generating document summary...")

summary = SummaryGenerator.generate(rendered_html)

logger.info("Building professional HTML document...")

html_document = HTMLDocumentBuilder.build(
title=title,
rendered_html=rendered_html,
toc_html=toc_html,
summary=summary
)

output_path.parent.mkdir(
parents=True,
exist_ok=True
)

logger.info("Rendering PDF...")

# Use the Markdown file's directory as the base URL
# so relative image paths continue to work.
HTML(
string=html_document,
base_url=str(markdown_path.parent.resolve())
).write_pdf(
str(output_path)
)

logger.info("PDF successfully generated!")
logger.info("Output: %s", output_path.resolve())

return output_path

COMMAND LINE INTERFACE

def parse_arguments():

parser = argparse.ArgumentParser(
prog="markdown-to-pdf",
description=(
"Convert Markdown files into professionally "
"formatted enterprise PDF documents."
)
)

parser.add_argument(
"-m",
"--markdown",
required=True,
help="Path to the input Markdown file."
)

parser.add_argument(
"-o",
"--output",
help="Optional output PDF path."
)

return parser.parse_args()

def main():

args = parse_arguments()

markdown_path = Path(args.markdown).expanduser().resolve()

if not markdown_path.exists():
logger.error(
"Input file does not exist: %s",
markdown_path
)
sys.exit(1)

if not markdown_path.is_file():
logger.error("Input path is not a file.")
sys.exit(1)

if markdown_path.suffix.lower() != ".md":
logger.error("Only .md files are supported.")
sys.exit(1)

if args.output:
output_path = Path(args.output).expanduser().resolve()
else:
DEFAULT_OUTPUT_DIR.mkdir(
parents=True,
exist_ok=True
)

output_path = (
DEFAULT_OUTPUT_DIR /
f"{markdown_path.stem}.pdf"
).resolve()

if output_path.suffix.lower() != ".pdf":
logger.error("Output file must have a .pdf extension.")
sys.exit(1)

if output_path == markdown_path:
logger.error("Input and output paths must be different.")
sys.exit(1)

try:

logger.info("=" * 65)
logger.info("%s v%s", APP_NAME, VERSION)
logger.info("=" * 65)

PDFConverter.convert(
markdown_path,
output_path
)

logger.info("=" * 65)
logger.info("CONVERSION COMPLETED SUCCESSFULLY")
logger.info("=" * 65)

except Exception as exc:

logger.exception("PDF conversion failed: %s", exc)
sys.exit(1)
